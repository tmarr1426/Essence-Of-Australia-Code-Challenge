Step 3: Create RegisterForm.tsx

Create:

src/components/auth/RegisterForm.tsx
import { useState } from "react";
import type { FormEvent } from "react";
import { useNavigate } from "react-router-dom";

import * as AuthService from "../../services/AuthService";
import { useAuth } from "../../hooks/useAuth";


export default function RegisterForm() {

    const [username, setUsername] = useState("");

    const [email, setEmail] = useState("");

    const [password, setPassword] = useState("");

    const [error, setError] = useState("");


    const { login: loginUser } = useAuth();

    const navigate = useNavigate();



    const handleSubmit = async (
        e: FormEvent<HTMLFormElement>
    ) => {

        e.preventDefault();


        try {

            const user = await AuthService.register(
                username,
                email,
                password
            );


            loginUser(user);


            navigate("/");


        } catch (err) {

            console.error(err);

            setError(
                "Unable to create account."
            );

        }

    };


    return (

        <form onSubmit={handleSubmit}>

            <input
                type="text"
                placeholder="Username"
                value={username}
                onChange={(e) =>
                    setUsername(e.target.value)
                }
            />


            <input
                type="email"
                placeholder="Email"
                value={email}
                onChange={(e) =>
                    setEmail(e.target.value)
                }
            />


            <input
                type="password"
                placeholder="Password"
                value={password}
                onChange={(e) =>
                    setPassword(e.target.value)
                }
            />


            <button type="submit">
                Register
            </button>


            {error && <p>{error}</p>}

        </form>

    );

}

The MySQL Docker image will correctly interpret:

/*!40000 ALTER TABLE `games` ENABLE KEYS */;

as:

ALTER TABLE `games` ENABLE KEYS;

The /*! ... */ syntax is a MySQL feature.

Good question. Since you are now using Docker Compose with:

mysql:
  environment:
    MYSQL_DATABASE: gamevoteapp

you do not need this anymore:

CREATE DATABASE IF NOT EXISTS GameVoteApp;
USE GameVoteApp;

The MySQL Docker image creates the database automatically during initialization.

What your Docker setup is doing

When MySQL starts for the first time, this:

MYSQL_DATABASE: gamevoteapp

is equivalent to running:

CREATE DATABASE gamevoteapp;

before it executes:

./database/schema.sql:/docker-entrypoint-initdb.d/init.sql

Your schema.sql is executed inside that database context.

Your schema should start like this instead

I would make the top of schema.sql:

SET FOREIGN_KEY_CHECKS = 0;

Then:

CREATE TABLE `users` (
...
);

Then:

CREATE TABLE `games` (
...
);

etc.

Should it be lowercase?

Yes, I recommend lowercase:

gamevoteapp

not:

GameVoteApp

Reason:

Windows MySQL is usually case-insensitive
Linux MySQL containers can be case-sensitive depending on configuration
Docker reviewers may run this on Linux

Keeping everything lowercase avoids headaches.

So make these match:

docker-compose.yml
mysql:
  environment:
    MYSQL_DATABASE: gamevoteapp
config.php
define("DB_NAME", "gamevoteapp");
schema.sql

Do not include:

CREATE DATABASE GameVoteApp;
USE GameVoteApp;
